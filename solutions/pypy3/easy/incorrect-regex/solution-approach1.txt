import re

if __name__ == '__main__':
    t = int(input())
    for _ in range(t):
        s = input()
        # HackerRank considers consecutive quantifier patterns invalid
        if any(bad in s for bad in ('*+', '++', '+*', '?+')):
            print(False)
            continue
        try:
            re.compile(s)
            print(True)
        except re.error:
            print(False)
