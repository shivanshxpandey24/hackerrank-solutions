def print_rangoli(size):
    chars = [chr(97 + i) for i in range(size)]
    width = 4 * size - 3
    lines = []
    
    for i in range(size):
        # Slice from outer letter inwards
        row_chars = chars[size - 1 : size - 1 - i : -1] + chars[size - 1 - i : size]
        lines.append('-'.join(row_chars).center(width, '-'))
        
    # Full pattern: top rows + bottom half (mirror of top rows except center)
    result = '\n'.join(lines + lines[-2::-1])
    print(result)
