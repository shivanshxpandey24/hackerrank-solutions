from collections import deque

def can_stack_cubes():
    t = int(input())
    for _ in range(t):
        n = int(input())
        d = deque(map(int, input().split()))
        
        last = float('inf')
        possible = True
        
        while d:
            # Pick the larger of the two ends
            if d[0] >= d[-1]:
                chosen = d.popleft()
            else:
                chosen = d.pop()
            
            # The chosen cube must be <= the previously placed base cube
            if chosen > last:
                possible = False
                break
            
            last = chosen
            
        print("Yes" if possible else "No")

if __name__ == '__main__':
    can_stack_cubes()
